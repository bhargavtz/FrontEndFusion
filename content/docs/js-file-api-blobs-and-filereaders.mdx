# File API, Blobs and FileReaders

A `File` is a special kind of `Blob` that carries a name and a last-modified date. Browser scripts cannot read arbitrary files from disk, but when a user picks files through an `input` element the page receives `File` objects it is allowed to inspect. Those objects expose `name`, `size`, `type` (a MIME string) and `lastModified`.

## Reading and slicing

`FileReader` reads a `Blob` asynchronously and reports through events. `readAsText(file)` decodes the bytes into a string, while `readAsDataURL(file)` produces a `data:` URL suitable for an image source or a download link. When reading finishes, `reader.result` holds the value and `reader.onload` fires; problems surface through `reader.onerror`. A file can be cut into smaller `Blob` pieces with `file.slice(start, end)`, which is the basis for chunked uploads or peeking at a header.

## Multiple files and downloads

Adding the `multiple` attribute to the input lets users choose several files at once, and `accept` filters the picker by extension or MIME type — for example `accept=".txt,.csv,text/plain"`. Loop over `input.files` to process each one in turn.

To create a file on the client, build a `Blob` from an array of strings, wrap it in a temporary URL with `URL.createObjectURL()`, and let an anchor with a `download` attribute save it. Release the URL with `URL.revokeObjectURL()` when finished. The demo below reads local files only and generates its own sample CSV, so it never contacts the network.

```html
<!doctype html>
<html lang="en">
  <head>
    <meta charset="utf-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1" />
    <title>File API explorer</title>
    <style>
      body { margin: 0; padding: 24px 16px; font-family: system-ui, sans-serif; color: #2a2f3c; background: linear-gradient(135deg, #f2f0fb, #eef7f5 80%); }
      .card { max-width: 560px; margin: 0 auto; padding: 22px; background: #fff; border-radius: 20px; box-shadow: 0 20px 42px -30px rgba(50, 40, 90, .5); }
      h1 { margin: 0 0 6px; font-size: 1.15rem; }
      .lead { margin: 0 0 14px; font-size: .84rem; color: #666b7e; }
      input[type=file] { width: 100%; padding: 10px; border: 1px dashed #bfb8dc; border-radius: 11px; background: #faf9ff; font-size: .78rem; }
      .row { display: flex; flex-wrap: wrap; gap: 8px; margin-top: 12px; }
      button { padding: 8px 11px; border: 0; border-radius: 9px; background: #5a4fa3; color: #fff; font: inherit; font-size: .76rem; cursor: pointer; }
      button:hover { background: #453c82; }
      pre { margin: 14px 0 0; min-height: 190px; padding: 12px; border-radius: 11px; background: #1f2030; color: #e0ddf7; font-size: .75rem; line-height: 1.6; white-space: pre-wrap; word-break: break-all; }
      .note { margin-top: 10px; font-size: .74rem; color: #9a4b4b; }
    </style>
  </head>
  <body>
    <div class="card">
      <h1>File API explorer</h1>
      <p class="lead">Pick a small text or CSV file, then try each operation.</p>
      <input id="file" type="file" multiple accept=".txt,.csv,text/plain,text/csv" />
      <div class="row">
        <button data-act="text">Read as text</button>
        <button data-act="url">Read as data URL</button>
        <button data-act="slice">Slice first 16 bytes</button>
        <button data-act="props">Show properties</button>
        <button data-act="csv">Download sample CSV</button>
      </div>
      <p class="note" id="warn"></p>
      <pre id="out">Choose a file above to begin.</pre>
    </div>
    <script>
      var out = document.getElementById("out");
      var input = document.getElementById("file");
      var warn = document.getElementById("warn");
      var hasReader = typeof FileReader !== "undefined";
      var hasBlob = typeof Blob !== "undefined";

      if (!hasReader || !hasBlob) {
        warn.textContent = "This browser lacks FileReader or Blob. The download demo is unavailable.";
      }

      function line(s) { out.textContent += s + "\n"; }
      function firstFile() { return input.files && input.files[0]; }

      function readAs(method, label) {
        var file = firstFile();
        if (!file) { line("No file selected."); return; }
        if (!hasReader) { line("FileReader unsupported."); return; }
        var reader = new FileReader();
        reader.onload = function () {
          var value = String(reader.result);
          line(label + " of " + file.name + " (" + value.length + " chars):");
          line(value.slice(0, 160) + (value.length > 160 ? " …" : ""));
        };
        reader.onerror = function () { line(label + " failed."); };
        if (method === "text") reader.readAsText(file);
        else reader.readAsDataURL(file);
      }

      document.querySelector(".row").addEventListener("click", function (e) {
        var btn = e.target.closest("button[data-act]");
        if (!btn) return;
        var act = btn.dataset.act;
        var file = firstFile();

        if (act === "text") readAs("text", "readAsText");
        if (act === "url") readAs("url", "readAsDataURL");

        if (act === "slice") {
          if (!file) { line("No file selected."); return; }
          var part = file.slice(0, 16);
          line("slice(0, 16) -> Blob of " + part.size + " bytes, type " + JSON.stringify(part.type));
          readSlice(part);
        }

        if (act === "props") {
          if (!file) { line("No file selected."); return; }
          line("files chosen: " + input.files.length);
          Array.prototype.forEach.call(input.files, function (f) {
            line(f.name + " | " + f.size + " bytes | type " + JSON.stringify(f.type) + " | modified " + new Date(f.lastModified).toLocaleString());
          });
        }

        if (act === "csv") {
          if (!hasBlob) { line("Blob unsupported."); return; }
          var csv = "name,score\nAda,95\nLin,88\n";
          var blob = new Blob([csv], { type: "text/csv" });
          var url = URL.createObjectURL(blob);
          var a = document.createElement("a");
          a.href = url; a.download = "sample.csv";
          document.body.appendChild(a); a.click(); a.remove();
          setTimeout(function () { URL.revokeObjectURL(url); }, 1000);
          line("Created a " + blob.size + "-byte CSV Blob and triggered a download.");
        }
      });

      function readSlice(blob) {
        if (!hasReader) return;
        var r = new FileReader();
        r.onload = function () { line("first bytes: " + JSON.stringify(String(r.result))); };
        r.readAsText(blob);
      }

      input.addEventListener("change", function () {
        out.textContent = "";
        line(input.files.length + " file(s) selected.");
      });
    </script>
  </body>
</html>
```
