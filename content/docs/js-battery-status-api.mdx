# Battery Status API

The Battery Status API exposes a device's power state to web pages through a `BatteryManager` object. You obtain it with `navigator.getBattery()`, which returns a promise that resolves to the manager. The information is deliberately coarse — a percentage and a few durations — but it is enough to adapt an interface, for example by dimming animations when power is low.

## Properties and events

A `BatteryManager` has four read-only properties. `level` is a number from `0` to `1`. `charging` is a boolean. `chargingTime` and `dischargingTime` give seconds until full or empty, using `Infinity` when the value is unknown. Each property has a matching event: `levelchange`, `chargingchange`, `chargingtimechange` and `dischargingtimechange`. Attach listeners with `addEventListener` to update the UI as the battery drains or a charger is plugged in.

## Privacy and fallbacks

Because battery level can be used for fingerprinting, several browsers have removed or restricted this API. Always check that `navigator.getBattery` is a function before calling it, and catch a rejected promise. When it is unavailable, the demo switches to a clearly labelled simulated battery with a slider and a charging toggle, so the event-driven model is still visible.

    <div class="card">
      <h1>Battery Status demo</h1>
      <p class="lead">Real API when available, a labelled simulation otherwise.</p>
      <span id="badge" class="badge">checking…</span>
      <div class="gauge"><div class="fill" id="fill"></div></div>
      <div class="label"><span id="pct">50%</span><span id="state">idle</span></div>
      <div class="sim" id="sim">

    <script>
      var out = document.getElementById("out");
      var badge = document.getElementById("badge");
      var fill = document.getElementById("fill");
      var pct = document.getElementById("pct");
      var state = document.getElementById("state");
      var sim = document.getElementById("sim");
      var level = document.getElementById("level");
      var charging = document.getElementById("charging");

      function line(s) { out.textContent += s + "\n"; }
      function time(seconds) {
        if (!isFinite(seconds) || seconds <= 0) return "unknown";
        var m = Math.round(seconds / 60);
        return m >= 60 ? Math.floor(m / 60) + "h " + (m % 60) + "m" : m + " min";
      }

      function render(levelValue, isCharging, chTime, disTime) {
        var p = Math.round(levelValue * 100);
        fill.style.width = p + "%";
        pct.textContent = p + "%";
        state.textContent = isCharging ? "charging · full in " + time(chTime) : "on battery · " + time(disTime) + " left";
      }

      var real = typeof navigator.getBattery === "function";
      badge.textContent = real ? "Real BatteryManager API" : "Simulated battery (API unavailable)";
      badge.style.background = real ? "#e9f2ec" : "#fdf1de";
      badge.style.color = real ? "#2c7a4f" : "#9a6a22";
      line(real ? "navigator.getBattery() detected." : "Battery Status API missing — using simulated values.");

      if (real) {
        sim.style.display = "none";
        navigator.getBattery().then(function (b) {
          function paint() { render(b.level, b.charging, b.chargingTime, b.dischargingTime); }
          paint();
          ["levelchange", "chargingchange", "chargingtimechange", "dischargingtimechange"].forEach(function (evt) {
            b.addEventListener(evt, function () {
              line("event " + evt + " -> level " + b.level.toFixed(2) + ", charging " + b.charging);
              paint();
            });
          });
          line("Listening for four battery change events.");
        }).catch(function (e) { line("getBattery() rejected: " + e.name); });
      } else {
        function fire(name) {
          line("event " + name + " -> level " + (level.value / 100).toFixed(2) + ", charging " + charging.checked);
          render(level.value / 100, charging.checked, (100 - level.value) * 60, level.value * 36);
        }
        level.addEventListener("input", function () { fire("levelchange"); });
        charging.addEventListener("change", function () { fire("chargingchange"); });
        render(0.5, false, 3000, 1800);
      }

      document.getElementById("clear").addEventListener("click", function () { out.textContent = ""; });
    </script>
  </body>
</html>
