# History

The `history` object is your window into the browser's session history — the list of places visited in the current tab. Single-page applications use it to change the address bar and respond to the Back and Forward buttons without reloading the whole page. Each entry stores a URL, a state object, and a title.

## Pushing and replacing entries

`history.pushState(state, title, url)` adds a new entry and moves the address bar to `url`, but does not load that page. The `title` argument is mostly ignored by browsers today. `history.replaceState()` is similar, except it updates the current entry instead of adding a new one — handy for refining a URL after a filter changes. The URL must be same-origin, or the call throws a security error.

```js
history.pushState({ page: "about" }, "", "/about");
history.replaceState({ page: "about", tab: 2 }, "", "/about?tab=2");
```

Since the browser does not fetch anything when you push a state, your own code must render the matching view.

## Moving and listening

Use `history.back()`, `history.forward()`, and `history.go(delta)` to move through the list; `go(0)` reloads the current entry. Navigation triggers a `popstate` event, which is where you read `event.state` and update the interface. Note that `pushState` and `replaceState` do **not** fire `popstate` — call your render function yourself after those calls.

Sandboxed frames with an opaque origin may reject history calls. The demo guards each call in a `try` block and keeps an in-memory stack as a fallback, so navigation stays visible even when the real API is unavailable.

```html
<!doctype html>
<html lang="en">
  <head>
    <meta charset="utf-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1" />
    <title>History navigator</title>
    <style>
      body {
        margin: 0;
        padding: 40px 18px;
        font-family: system-ui, sans-serif;
        background: #eef3fb;
        color: #1c2c4a;
      }
      .card {
        max-width: 560px;
        margin: 0 auto;
        padding: 28px 30px 30px;
        background: #fff;
        border-radius: 20px;
        box-shadow: 0 26px 54px -40px rgba(28, 44, 74, 0.6);
      }
      h1 { margin: 0 0 6px; font-size: 1.25rem; }
      p { margin: 0 0 18px; color: #66789b; font-size: 0.9rem; }
      .nav { display: flex; gap: 8px; flex-wrap: wrap; margin-bottom: 18px; }
      button {
        border: 0; border-radius: 10px; padding: 9px 14px; cursor: pointer;
        font: inherit; font-size: 0.83rem; font-weight: 600;
        background: #3a63d6; color: #fff;
      }
      button.alt { background: #e4ebfb; color: #34509e; }
      .view {
        padding: 22px; border-radius: 14px; background: #f4f7ff;
        min-height: 90px; margin-bottom: 14px;
      }
      .view h2 { margin: 0 0 6px; font-size: 1.05rem; }
      .view p { margin: 0; color: #5e7096; }
      .out {
        margin: 0; padding: 12px 14px; background: #172444;
        color: #a9bdf0; border-radius: 10px; font-size: 0.8rem; line-height: 1.7;
      }
    </style>
  </head>
  <body>
    <div class="card">
      <h1>History navigator</h1>
      <p>Pages change without a reload. Use Back and Forward just like a real site.</p>
      <div class="nav">
        <button data-go="home">Home</button>
        <button data-go="about">About</button>
        <button data-go="contact">Contact</button>
        <button id="back" class="alt">Back</button>
        <button id="forward" class="alt">Forward</button>
        <button id="replace" class="alt">Replace URL</button>
      </div>
      <div class="view" id="view"></div>
      <pre id="out" class="out">Current page: home</pre>
    </div>
    <script>
      var pages = {
        home: ["Home", "Welcome to the demo."],
        about: ["About", "This view was pushed, not reloaded."],
        contact: ["Contact", "Reach us at hello@example.test."]
      };
      var view = document.getElementById("view");
      var out = document.getElementById("out");
      var stack = ["home"];
      var index = 0;
      var realHistory = true;

      function render(name) {
        var p = pages[name] || pages.home;
        view.innerHTML = "";
        var h = document.createElement("h2");
        h.textContent = p[0];
        var t = document.createElement("p");
        t.textContent = p[1];
        view.appendChild(h);
        view.appendChild(t);
        out.textContent = "Current page: " + name + "\nStack depth: " + (index + 1);
      }
      function go(name) {
        index = stack.length;
        stack.push(name);
        index = stack.length - 1;
        try {
          history.pushState({ page: name }, "", "#" + name);
        } catch (e) {
          realHistory = false;
        }
        render(name);
      }
      window.addEventListener("popstate", function (e) {
        var name = (e.state && e.state.page) || "home";
        render(name);
      });
      Array.prototype.forEach.call(document.querySelectorAll("button[data-go]"), function (b) {
        b.onclick = function () { go(b.getAttribute("data-go")); };
      });
      document.getElementById("back").onclick = function () {
        if (index > 0) { index--; render(stack[index]); try { history.back(); } catch (e) {} }
      };
      document.getElementById("forward").onclick = function () {
        if (index < stack.length - 1) { index++; render(stack[index]); try { history.forward(); } catch (e) {} }
      };
      document.getElementById("replace").onclick = function () {
        try { history.replaceState({ page: stack[index] }, "", "#" + stack[index]); }
        catch (e) { realHistory = false; }
        out.textContent += "\nURL replaced" + (realHistory ? "." : " (fallback mode).");
      };
      render("home");
    </script>
  </body>
</html>
```

Notice that `pushState()` does not fire `popstate`, which is why the button handlers call `render()` directly, while only the Back and Forward paths rely on the event.
