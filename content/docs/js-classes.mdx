# Classes

Class syntax is a tidy layer over the constructor and prototype system. It gathers the setup code, the shared methods, and the computed properties of a type into one readable block, while keeping the underlying prototype chain intact.

## A blueprint with a constructor

The `constructor` method runs when you call the class with `new`. Arguments become instance fields, usually assigned to `this`, and every method written inside the class body ends up on the prototype. Because of that, two instances hold their own data but reach the same function objects.

Getters and setters use `get` and `set` in front of a method name. They look like ordinary fields from the outside, yet they run code on read and write. A setter is a good place to validate input or round a value before it is stored.

## Static members, subclasses, and privacy

A method marked `static` belongs to the class itself rather than to instances. Factories and constant tables are common uses. A subclass uses `extends` to inherit, calls `super()` in its constructor, and can call `super.method()` to reuse a parent implementation before adding its own detail.

For private data, modern engines offer hash-prefixed fields. A widely compatible alternative is a `WeakMap` kept outside the class: each instance becomes a key, and the stored object is reachable only through code that holds the map. Nothing on the instance exposes the secret.

Methods declared in a class body are non-enumerable, unlike properties assigned one by one inside the constructor. They do not appear in `for...in` loops or `Object.keys()`, which keeps enumeration and serialization focused on the data. It is a small detail, but it explains why class instances stringify so cleanly.

<script>
  var secretBatch = new WeakMap();

  class Pastry {
    constructor(name, price) {
      this.name = name;
      this._price = price;
      secretBatch.set(this, { code: "B-" + Math.floor(100 + Math.random() * 900) });
    }
    get price() { return this._price; }
    set price(value) {
      if (typeof value !== "number" || value <= 0) {
        throw new Error("Price must be a positive number");
      }
      this._price = Math.round(value * 100) / 100;
    }
    get batch() { return secretBatch.get(this).code; }
    describe() { return this.name + " - $" + this.price.toFixed(2); }
    static fromLabel(label) {
      var parts = label.split(":");
      return new Pastry(parts[0], Number(parts[1]));
    }
  }

  class Cake extends Pastry {
    constructor(name, price, layers) {
      super(name, price);
      this.layers = layers;
    }
    describe() {
      return super.describe() + " (" + this.layers + " layers)";
    }
  }

  var items = [];
  var list = document.getElementById("items");
  var out = document.getElementById("out");

  function render() {
    list.innerHTML = "";
    var total = 0;
    items.forEach(function (item) {
      total += item.price;
      var card = document.createElement("div");
      card.className = "item";
      card.textContent = item.describe() + "   batch " + item.batch;
      list.appendChild(card);
    });
    out.textContent = items.length
      ? items.length + " item(s), total $" + total.toFixed(2) + ". Batch codes are held in a WeakMap."
      : "Order empty. Add something tasty.";
  }

  document.getElementById("addCroissant").addEventListener("click", function () {
    items.push(new Pastry("Croissant", 3.5));
    render();
  });
  document.getElementById("addCake").addEventListener("click", function () {
    items.push(new Cake("Carrot cake", 6.25, 3));
    render();
  });
  document.getElementById("addLabel").addEventListener("click", function () {
    items.push(Pastry.fromLabel("Cinnamon roll:4.25"));
    render();
  });
  document.getElementById("clear").addEventListener("click", function () {
    items = [];
    render();
  });

  render();
</script>
