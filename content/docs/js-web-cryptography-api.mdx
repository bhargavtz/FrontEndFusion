# Web Cryptography API

The Web Cryptography API gives scripts access to cryptographic primitives that are fast and reviewed by the browser vendor. Everything lives under `crypto.subtle`, and almost every operation is asynchronous and returns a promise. The API only works in secure contexts, which means HTTPS or localhost, so a sandboxed preview may need to fall back to an explanatory message.

## Digests and randomness

`crypto.subtle.digest("SHA-256", data)` takes an `ArrayBuffer` (or typed array) and resolves to an `ArrayBuffer` containing the hash. To display it, wrap the buffer in a `Uint8Array` and format each byte as two hexadecimal digits. A digest is one-way: the same input always produces the same output, while changing a single character changes it completely.

For random data, use `crypto.getRandomValues(typedArray)`, which fills the array with cryptographically strong bytes. It is synchronous and is the right choice for tokens, salts and nonces — `Math.random()` is not suitable for security purposes.

## Keys and PEM

`crypto.subtle.generateKey({ name: "RSASSA-PKCS1-v1_5", modulusLength: 2048, publicExponent: new Uint8Array([1, 0, 1]), hash: "SHA-256" }, true, ["sign", "verify"])` produces an RSA key pair. With `extractable` set to `true`, `exportKey("spki", publicKey)` and `exportKey("pkcs8", privateKey)` return buffers that can be base64-encoded into the familiar PEM format. `importKey()` reverses the process. The demo exercises hashing, randomness and a full export/import round trip, with feature detection and a visible fallback.

    <div class="card">
      <h1>Web Crypto lab</h1>
      <p class="lead">Hashing, randomness and an RSA export/import round trip.</p>
      <span id="badge" class="badge">checking…</span>
      <input id="text" value="hello crypto" aria-label="text to hash" />
      <div class="row">
        <button data-act="hash">SHA-256 digest</button>

    <script>
      var out = document.getElementById("out");
      var badge = document.getElementById("badge");
      var subtle = window.crypto && window.crypto.subtle;
      var hasSubtle = !!(subtle && typeof subtle.digest === "function");

      function line(s) { out.textContent += s + "\n"; }
      function hex(buf) {
        return Array.prototype.map.call(new Uint8Array(buf), function (b) {
          return ("0" + b.toString(16)).slice(-2);
        }).join("");
      }
      function toBuffer(str) {
        if (window.TextEncoder) return new TextEncoder().encode(str);
        var a = new Uint8Array(str.length);
        for (var i = 0; i < str.length; i++) a[i] = str.charCodeAt(i) & 255;
        return a;
      }
      function bufToB64(buf) {
        var bytes = new Uint8Array(buf), s = "";
        for (var i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]);
        return btoa(s);
      }
      function toPem(buf, label) {
        var b64 = bufToB64(buf);
        var wrapped = b64.match(/.{1,64}/g).join("\n");
        return "-----BEGIN " + label + "-----\n" + wrapped + "\n-----END " + label + "-----";
      }
      function pemToBuffer(pem) {
        var b64 = pem.replace(/-----[^-]+-----/g, "").replace(/\s+/g, "");
        var bin = atob(b64);
        var a = new Uint8Array(bin.length);
        for (var i = 0; i < bin.length; i++) a[i] = bin.charCodeAt(i);
        return a.buffer;
      }

      badge.textContent = hasSubtle ? "SubtleCrypto available" : "SubtleCrypto unavailable — secure context required";
      line(hasSubtle ? "crypto.subtle is ready." : "This sandbox does not expose crypto.subtle; cryptographic steps cannot run here.");

      document.querySelector(".row").addEventListener("click", function (e) {
        var btn = e.target.closest("button[data-act]");
        if (!btn) return;
        var act = btn.dataset.act;

        if (act === "hash") {
          var text = document.getElementById("text").value;
          if (!hasSubtle) { line("SHA-256 skipped: SubtleCrypto unavailable."); return; }
          subtle.digest("SHA-256", toBuffer(text)).then(function (buf) {
            line("SHA-256(" + JSON.stringify(text) + ") = " + hex(buf));
          }).catch(function (err) { line("digest failed: " + err.name); });
        }

        if (act === "rand") {
          if (!(window.crypto && crypto.getRandomValues)) { line("getRandomValues unavailable."); return; }
          var bytes = new Uint8Array(16);
          crypto.getRandomValues(bytes);
          line("16 random bytes = " + hex(bytes.buffer));
        }

        if (act === "rsa") {
          if (!hasSubtle) { line("RSA demo skipped: SubtleCrypto unavailable."); return; }
          line("Generating a 2048-bit RSA key pair…");
          subtle.generateKey({
            name: "RSASSA-PKCS1-v1_5",
            modulusLength: 2048,
            publicExponent: new Uint8Array([1, 0, 1]),
            hash: "SHA-256"
          }, true, ["sign", "verify"]).then(function (pair) {
            return Promise.all([
              subtle.exportKey("spki", pair.publicKey),
              subtle.exportKey("pkcs8", pair.privateKey)
            ]).then(function (keys) {
              var pubPem = toPem(keys[0], "PUBLIC KEY");
              var privPem = toPem(keys[1], "PRIVATE KEY");
              line("public PEM (first 100 chars): " + pubPem.slice(0, 100) + "…");
              line("private PEM length: " + privPem.length + " characters");
              return subtle.importKey("spki", pemToBuffer(pubPem), { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" }, true, ["verify"]);
            }).then(function (key) {
              line("re-imported key: type=" + key.type + ", algorithm=" + key.algorithm.name);
            });
          }).catch(function (err) { line("RSA step failed: " + err.name + " — " + err.message); });
        }
      });
    </script>
  </body>
</html>
